## 3. ПРОГРАММНАЯ РЕАЛИЗАЦИЯ ИНТЕГРАЦИИ

### 3.1. Клиентская динамическая валидация формы подачи заявки (`apply.html`)

Для предотвращения некорректных отправок и мгновенной обратной связи внедрен клиентский скрипт, сопоставляющий data-атрибуты выбранного студента и темы:

```javascript
document.addEventListener("DOMContentLoaded", function() {
    const studentSelect = document.getElementById("studentSelect");
    const topicSelect = document.getElementById("topicSelect");
    const valCard = document.getElementById("validationCard");
    const checkCourse = document.getElementById("checkCourse");
    const checkGpa = document.getElementById("checkGpa");
    const checkQuota = document.getElementById("checkQuota");
    const submitBtn = document.getElementById("submitBtn");

    function validateSelection() {
        const studentOpt = studentSelect.options[studentSelect.selectedIndex];
        const topicOpt = topicSelect.options[topicSelect.selectedIndex];

        if (!studentOpt.value || !topicOpt.value) {
            valCard.style.display = "none";
            submitBtn.disabled = false;
            return;
        }

        const studentCourse = parseInt(studentOpt.dataset.course);
        const studentGpa = parseFloat(studentOpt.dataset.gpa);
        const minCourse = parseInt(topicOpt.dataset.minCourse);
        const minGpa = parseFloat(topicOpt.dataset.minGpa);
        const capacity = parseInt(topicOpt.dataset.capacity);
        const current = parseInt(topicOpt.dataset.current);

        valCard.style.display = "block";

        const courseOk = studentCourse >= minCourse;
        const gpaOk = studentGpa >= minGpa;
        const quotaOk = current < capacity;

        checkCourse.innerHTML = courseOk
            ? `<span class="text-success">Курс соискателя (${studentCourse}) удовлетворяет требованию (>= ${minCourse})</span>`
            : `<span class="text-danger">Курс соискателя (${studentCourse}) ниже требуемого (>= ${minCourse})</span>`;

        checkGpa.innerHTML = gpaOk
            ? `<span class="text-success">Средний балл GPA (${studentGpa.toFixed(2)}) удовлетворяет требованию (>= ${minGpa.toFixed(2)})</span>`
            : `<span class="text-danger">Средний балл GPA (${studentGpa.toFixed(2)}) ниже порогового (>= ${minGpa.toFixed(2)})</span>`;

        checkQuota.innerHTML = quotaOk
            ? `<span class="text-success">Свободные квоты доступны (${current} / ${capacity} занято)</span>`
            : `<span class="text-danger">Квота мест полностью исчерпана (${current} / ${capacity})</span>`;

        submitBtn.disabled = !(courseOk && gpaOk && quotaOk);
    }

    studentSelect.addEventListener("change", validateSelection);
    topicSelect.addEventListener("change", validateSelection);
});
```

### 3.2. Контроллеры представлений Django (`topics/views.py`)

```python
def apply_view(request):
    """Контроллер обработки подачи заявки: двухфакторная валидация."""
    if request.method == "POST":
        student_id = request.POST.get("student")
        topic_id = request.POST.get("topic")

        student = get_object_or_404(Student, pk=student_id)
        topic = get_object_or_404(Topic, pk=topic_id)

        # Серверная проверка дублирования
        if Application.objects.filter(student=student, topic=topic, is_cancelled=False).exists():
            messages.error(request, f"Студент {student.name} уже подал активную заявку на данную тему.")
            return redirect("apply")

        # Серверная проверка допуска
        if not topic.is_suitable_for(student.course, student.gpa):
            messages.error(request, f"Студент {student.name} не соответствует академическим требованиям темы.")
            return redirect("apply")

        # Атомарное резервирование слота
        if not topic.reserve_slot():
            messages.error(request, f"На тему «{topic.title}» исчерпана квота свободных мест.")
            return redirect("apply")

        app = Application.objects.create(
            topic=topic,
            student=student,
            priority=student.calculate_priority(),
            is_cancelled=False,
        )

        messages.success(request, f"Заявка №{app.id} успешно подана и утверждена! Приоритет: {app.priority} баллов.")
        return redirect("applications_list")

    students = Student.objects.all().order_by("name")
    topics = Topic.objects.filter(is_available=True).order_by("title")
    return render(request, "topics/apply.html", {"students": students, "topics": topics})
```
