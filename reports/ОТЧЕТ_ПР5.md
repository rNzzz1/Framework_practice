   - `GET /api/topics/` — каталог тем курсовых работ с поддержкой фильтрации;

## 2. АРХИТЕКТУРА СЕРВЕРНОЙ ЧАСТИ И СХЕМА ДАННЫХ

В рамках практической работы осуществлен переход от консольного приложения с плоскими JSON-файлами к масштабируемой трехуровневой MVC/MVT архитектуре на базе веб-фреймворка Django.

### Структура базы данных и связи сущностей:
- **Discipline (Дисциплина):** справочник академических предметов кафедры. Связь 1:N с темами курсовых работ.
- **Student (Студент-соискатель):** обучающийся кафедры с атрибутами курса, среднего балла (GPA), признака бюджетной формы обучения и корпоративной электронной почты.
- **Topic (Тема курсовой):** предлагаемая кафедрой курсовая работа с указанием руководителя, пороговых требований (минимальный курс, минимальный GPA), лимита квоты мест (`capacity`) и текущего числа закрепленных студентов (`current_students`).
- **Application (Заявка):** факт выбора темы студентом с фиксацией даты подачи, рассчитанного рейтингового балла приоритета и статуса активности (`is_cancelled`).

## 3. ПРОГРАММНАЯ РЕАЛИЗАЦИЯ СЕРВЕРНОЙ ЛОГИКИ

### 3.1. Модели предметной области (`topics/models.py`)

```python
from django.db import models
from django.core.exceptions import ValidationError


class Discipline(models.Model):
    name = models.CharField(max_length=200, verbose_name="Наименование дисциплины", unique=True)
    code = models.CharField(max_length=50, verbose_name="Код дисциплины", blank=True)
    department = models.CharField(max_length=200, verbose_name="Кафедра", default="Кафедра индустриального программирования")

    class Meta:
        verbose_name = "Дисциплина"
        verbose_name_plural = "Дисциплины"
        ordering = ["name"]

    def __str__(self) -> str:
        return self.name


class Student(models.Model):
    name = models.CharField(max_length=150, verbose_name="ФИО студента")
    course = models.PositiveSmallIntegerField(verbose_name="Курс обучения", default=3)
    gpa = models.FloatField(verbose_name="Средний балл (GPA)", default=4.0)
    is_budget = models.BooleanField(verbose_name="Бюджетная форма обучения", default=True)
    email = models.EmailField(verbose_name="Электронная почта", unique=True)

    class Meta:
        verbose_name = "Студент"
        verbose_name_plural = "Студенты"
        ordering = ["name"]

    def calculate_priority(self) -> int:
        """Расчет приоритета заявки: base = GPA * 20, бонус за бюджет = 10 баллов."""
        base = int(float(self.gpa) * 20)
        bonus = 10 if self.is_budget else 0
        return base + bonus


class Topic(models.Model):
    title = models.CharField(max_length=255, verbose_name="Название темы")
    discipline = models.ForeignKey(Discipline, on_delete=models.CASCADE, related_name="topics", verbose_name="Дисциплина")
    supervisor = models.CharField(max_length=150, verbose_name="Научный руководитель")
    min_course = models.PositiveSmallIntegerField(verbose_name="Минимальный курс", default=3)
    min_gpa = models.FloatField(verbose_name="Минимальный балл GPA", default=3.5)
    capacity = models.PositiveSmallIntegerField(verbose_name="Квота мест", default=1)
    current_students = models.PositiveSmallIntegerField(verbose_name="Записано студентов", default=0)
    is_available = models.BooleanField(verbose_name="Доступна для выбора", default=True)

    def is_suitable_for(self, student_course: int, gpa: float) -> bool:
        return (student_course >= self.min_course) and (gpa >= self.min_gpa)

    def has_free_capacity(self) -> bool:
        return self.current_students < self.capacity

    def reserve_slot(self) -> bool:
        if not self.has_free_capacity() or not self.is_available:
            return False
        self.current_students += 1
        if self.current_students >= self.capacity:
            self.is_available = False
        self.save()
        return True

    def release_slot(self) -> None:
        self.current_students = max(0, self.current_students - 1)
        self.is_available = True
        self.save()


class Application(models.Model):
    topic = models.ForeignKey(Topic, on_delete=models.CASCADE, related_name="applications")
    student = models.ForeignKey(Student, on_delete=models.CASCADE, related_name="applications")
    application_date = models.DateField(auto_now_add=True)
    priority = models.IntegerField(default=0)
    is_cancelled = models.BooleanField(default=False)

    @property
    def status(self) -> str:
        return "ОТМЕНЕНА" if self.is_cancelled else "ОДОБРЕНА"

    def cancel(self) -> None:
        if not self.is_cancelled:
            self.is_cancelled = True
            self.save()
            self.topic.release_slot()
```

### 3.2. RESTful JSON API контроллеры (`topics/views.py`)

```python
import json
from django.http import JsonResponse, HttpResponseBadRequest
from django.shortcuts import get_object_or_404
from django.views.decorators.http import require_POST
from django.views.decorators.csrf import csrf_exempt
from .models import Topic, Student, Application


def api_topics_list(request):
    """GET /api/topics/ — выдача списка тем курсовых работ."""
    topics = Topic.objects.select_related("discipline").all()
    data = [{
        "id": t.id,
        "title": t.title,
        "discipline": t.discipline.name,
        "supervisor": t.supervisor,
        "min_course": t.min_course,
        "min_gpa": t.min_gpa,
        "capacity": t.capacity,
        "current_students": t.current_students,
        "is_available": t.is_available,
    } for t in topics]
    return JsonResponse({"topics": data, "count": len(data)})


@csrf_exempt
@require_POST
def api_create_application(request):
    """POST /api/applications/create/ — подача заявки через JSON API."""
    try:
        payload = json.loads(request.body)
        student_id = payload.get("student_id")
        topic_id = payload.get("topic_id")
    except (json.JSONDecodeError, ValueError):
        return HttpResponseBadRequest(JsonResponse({"error": "Некорректный JSON"}))

    student = get_object_or_404(Student, pk=student_id)
    topic = get_object_or_404(Topic, pk=topic_id)

    if Application.objects.filter(student=student, topic=topic, is_cancelled=False).exists():
        return JsonResponse({"success": False, "error": "Студент уже подал активную заявку на эту тему."}, status=400)

    if not topic.is_suitable_for(student.course, student.gpa):
        return JsonResponse({"success": False, "error": "Студент не соответствует требованиям темы."}, status=400)

    if not topic.reserve_slot():
        return JsonResponse({"success": False, "error": "Квота мест на тему исчерпана."}, status=400)

    app = Application.objects.create(
        topic=topic,
        student=student,
        priority=student.calculate_priority(),
    )

    return JsonResponse({
        "success": True,
        "application_id": app.id,
        "status": app.status,
        "priority": app.priority,
    }, status=201)
```
