**Цель работы:**

```
[ Пользовательский клиент (Браузер / API) ]

|  - Маршрутизация URL, генерация представлений             |
```

## 3. ПРОГРАММНАЯ РЕАЛИЗАЦИЯ ИНФРАСТРУКТУРЫ

### 3.1. Спецификация Dockerfile

```dockerfile
FROM python:3.12-slim

ENV PYTHONDONTWRITEBYTECODE=1 \
    PYTHONUNBUFFERED=1 \
    PORT=8000

RUN apt-get update && apt-get install -y --no-install-recommends \
    curl \
    && rm -rf /var/lib/apt/lists/*

WORKDIR /app

COPY requirements.txt /app/
RUN pip install --no-cache-dir -r requirements.txt

COPY . /app/
RUN python manage.py collectstatic --noinput || true

EXPOSE 8000

HEALTHCHECK --interval=30s --timeout=5s --start-period=10s --retries=3 \
    CMD curl -f http://localhost:8000/api/health/ || exit 1

CMD ["gunicorn", "coursework_service.wsgi:application", "--bind", "0.0.0.0:8000", "--workers", "3", "--timeout", "60"]
```

### 3.2. Конфигурация веб-сервера Nginx (`nginx.conf`)

```nginx
events {
    worker_connections 1024;
}

http {
    include /etc/nginx/mime.types;
    default_type application/octet-stream;
    sendfile on;
    gzip on;
    gzip_comp_level 6;
    gzip_types text/plain text/css application/json application/javascript image/svg+xml;

    upstream django_app {
        server web:8000;
    }

    server {
        listen 80;
        server_name localhost 127.0.0.1;

        location /static/ {
            alias /var/html/static/;
            expires 30d;
        }

        location / {
            proxy_pass http://django_app;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        }

        location /api/health/ {
            proxy_pass http://django_app/api/health/;
            access_log off;
        }
    }
}
```

### 3.3. Оркестрация сервисов (`docker-compose.yml`)

```yaml
version: '3.8'

services:
  web:
    build: .
    container_name: coursework_service_web
    restart: always
    environment:
      - DJANGO_SETTINGS_MODULE=coursework_service.settings
      - PYTHONUNBUFFERED=1
    volumes:
      - ./data:/app/data
      - db_data:/app
      - static_volume:/app/staticfiles
    expose:
      - "8000"
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:8000/api/health/"]
      interval: 15s
      timeout: 5s
      retries: 3
      start_period: 10s

  nginx:
    image: nginx:1.25-alpine
    container_name: coursework_service_nginx
    restart: always
    ports:
      - "80:80"
    volumes:
      - ./nginx.conf:/etc/nginx/nginx.conf:ro
      - static_volume:/var/html/static:ro
    depends_on:
      web:
        condition: service_healthy

volumes:
  db_data:
  static_volume:
```

### 3.4. Эндпоинт мониторинга состояния (Health Check)
