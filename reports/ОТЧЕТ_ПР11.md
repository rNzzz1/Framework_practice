### 4.1. Исходный код модуля `topics/forms.py`
```python
from django import forms
from django.core.exceptions import ValidationError
from .models import Discipline, Student, Topic, Application


class TopicFilterForm(forms.Form):
    """Форма фильтрации и поиска каталога тем курсовых работ (GET-запросы)."""

    q = forms.CharField(
        required=False,
        label="Поисковый запрос",
        widget=forms.TextInput(attrs={"class": "form-control", "placeholder": "Поиск по названию темы, руководителю..."}),
    )
    discipline = forms.ModelChoiceField(
        queryset=Discipline.objects.all(),
        required=False,
        empty_label="Все дисциплины",
        label="Дисциплина",
        widget=forms.Select(attrs={"class": "form-select"}),
    )
    min_course = forms.ChoiceField(
        choices=[("", "Любой курс"), ("1", "1 курс"), ("2", "2 курс"), ("3", "3 курс"), ("4", "4 курс")],
        required=False,
        label="Курс обучения",
        widget=forms.Select(attrs={"class": "form-select"}),
    )
    sort = forms.ChoiceField(
        choices=[
            ("title", "По названию (А-Я)"),
            ("-title", "По названию (Я-А)"),
            ("min_gpa", "По минимальному GPA (по возрастанию)"),
            ("-min_gpa", "По минимальному GPA (по убыванию)"),
            ("capacity", "По квоте мест (по возрастанию)"),
            ("-capacity", "По квоте мест (по убыванию)"),
        ],
        required=False,
        label="Сортировка",
        widget=forms.Select(attrs={"class": "form-select"}),
    )


class ApplicationForm(forms.ModelForm):
    """Форма подачи новой заявки на закрепление темы (ModelForm)."""

    class Meta:
        model = Application
        fields = ["student", "topic"]
        widgets = {
            "student": forms.Select(attrs={"class": "form-select"}),
            "topic": forms.Select(attrs={"class": "form-select"}),
        }
        labels = {
            "student": "Студент-соискатель",
            "topic": "Тема курсовой работы",
        }

    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self.fields["topic"].queryset = Topic.objects.filter(is_available=True).order_by("title")
        self.fields["student"].queryset = Student.objects.all().order_by("name")

    def clean(self):
        """Комплексная бизнес-валидация подачи заявки."""
        cleaned_data = super().clean()
        student = cleaned_data.get("student")
        topic = cleaned_data.get("topic")

        if not student or not topic:
            return cleaned_data

        # 1. Запрет повторной активной заявки
        existing_active = Application.objects.filter(student=student, topic=topic, status=Application.Status.ACTIVE)
        if existing_active.exists():
            raise ValidationError(
                f"Студент «{student.name}» уже имеет активную утвержденную заявку на тему «{topic.title}»."
            )

        # 2. Проверка академического допуска студента (курс и GPA)
        if not topic.is_suitable_for(student.course, student.gpa):
            raise ValidationError(
                f"Студент {student.name} не соответствует академическим требованиям темы: "
                f"требуется курс >= {topic.min_course} (у студента {student.course}), "
                f"GPA >= {topic.min_gpa:.2f} (у студента {student.gpa:.2f})."
            )

        # 3. Проверка квоты свободных мест
        if not topic.has_free_capacity() or not topic.is_available:
            raise ValidationError(
                f"На тему «{topic.title}» исчерпана квота свободных мест ({topic.current_students}/{topic.capacity})."
            )

        return cleaned_data


class ApplicationEditForm(forms.ModelForm):
    """Форма редактирования заявки с учетом параметра instance."""

    class Meta:
        model = Application
        fields = ["student", "topic", "status"]
        widgets = {
            "student": forms.Select(attrs={"class": "form-select"}),
            "topic": forms.Select(attrs={"class": "form-select"}),
            "status": forms.Select(attrs={"class": "form-select"}),
        }
        labels = {
            "student": "Студент-соискатель",
            "topic": "Тема курсовой работы",
            "status": "Статус заявки",
        }

    def clean(self):
        cleaned_data = super().clean()
        student = cleaned_data.get("student")
        topic = cleaned_data.get("topic")
        status = cleaned_data.get("status")

        if not student or not topic:
            return cleaned_data

        # Исключаем текущую запись из поиска дубликатов
        query = Application.objects.filter(student=student, topic=topic, status=Application.Status.ACTIVE)
        if self.instance and self.instance.pk:
            query = query.exclude(pk=self.instance.pk)

        if status == Application.Status.ACTIVE and query.exists():
            raise ValidationError(f"Студент «{student.name}» уже имеет другую активную заявку на тему «{topic.title}».")

        if status == Application.Status.ACTIVE and not topic.is_suitable_for(student.course, student.gpa):
            raise ValidationError(
                f"Студент {student.name} не удовлетворяет требованиям темы: "
                f"курс {student.course} < {topic.min_course} или GPA {student.gpa:.2f} < {topic.min_gpa:.2f}."
            )

        return cleaned_data


class TopicForm(forms.ModelForm):
    """Форма создания/редактирования тем с загрузкой файлов (ImageField)."""

    class Meta:
        model = Topic
        fields = ["title", "discipline", "supervisor", "min_course", "min_gpa", "capacity", "is_available", "image"]
        widgets = {
            "title": forms.TextInput(attrs={"class": "form-control", "placeholder": "Введите название темы"}),
            "discipline": forms.Select(attrs={"class": "form-select"}),
            "supervisor": forms.TextInput(attrs={"class": "form-control", "placeholder": "ФИО руководителя, должность"}),
            "min_course": forms.NumberInput(attrs={"class": "form-control", "min": 1, "max": 6}),
            "min_gpa": forms.NumberInput(attrs={"class": "form-control", "step": "0.1", "min": "2.0", "max": "5.0"}),
            "capacity": forms.NumberInput(attrs={"class": "form-control", "min": 1, "max": 10}),
            "is_available": forms.CheckboxInput(attrs={"class": "form-check-input"}),
            "image": forms.ClearableFileInput(attrs={"class": "form-control"}),
        }

    def clean_capacity(self):
        capacity = self.cleaned_data.get("capacity")
        if capacity is not None and capacity < 1:
            raise ValidationError("Квота мест темы должна быть не менее 1.")
        return capacity

    def clean_min_gpa(self):
        min_gpa = self.cleaned_data.get("min_gpa")
        if min_gpa is not None and (min_gpa < 2.0 or min_gpa > 5.0):
            raise ValidationError("Минимальный средний балл должен быть в диапазоне от 2.0 до 5.0.")
        return min_gpa
```

### 4.2. Обработка форм в контроллерах (`topics/views.py`)

```python
def apply_view(request):
    """Контроллер подачи заявки: обработка GET и POST."""
    initial = {}
    topic_id = request.GET.get("topic_id")
    if topic_id:
        try:
            initial["topic"] = Topic.objects.get(pk=topic_id)
        except Topic.DoesNotExist:
            pass

    if request.method == "POST":
        form = ApplicationForm(request.POST)
        if form.is_valid():
            topic = form.cleaned_data["topic"]
            if not topic.reserve_slot():
                messages.error(request, f"Квота свободных мест на тему «{topic.title}» была исчерпана.")
                return render(request, "topics/apply.html", {"form": form})

            app = form.save()
            messages.success(request, f"Заявка №{app.id} успешно подана и утверждена!")
            return redirect("applications_list")
        else:
            messages.error(request, "Пожалуйста, исправьте ошибки при заполнении формы заявки.")
    else:
        form = ApplicationForm(initial=initial)

    return render(request, "topics/apply.html", {"form": form})


def application_edit_view(request, pk):
    """Контроллер изменения существующей заявки с instance."""
    app = get_object_or_404(Application, pk=pk)
    old_topic = app.topic
    old_status = app.status

    if request.method == "POST":
        form = ApplicationEditForm(request.POST, instance=app)
        if form.is_valid():
            new_app = form.save(commit=False)
            # Корректировка занятых мест при смене темы или статуса
            if old_topic != new_app.topic:
                if old_status == Application.Status.ACTIVE:
                    old_topic.release_slot()
                if new_app.status == Application.Status.ACTIVE:
                    new_app.topic.reserve_slot()
            elif old_status != new_app.status:
                if new_app.status == Application.Status.CANCELLED:
                    new_app.topic.release_slot()
                elif old_status == Application.Status.CANCELLED and new_app.status == Application.Status.ACTIVE:
                    new_app.topic.reserve_slot()

            new_app.save()
            messages.success(request, f"Заявка №{new_app.id} успешно обновлена.")
            return redirect("applications_list")
        else:
            messages.error(request, "Ошибка валидации при редактировании заявки.")
    else:
        form = ApplicationEditForm(instance=app)

    return render(request, "topics/application_edit.html", {"form": form, "application": app})
```

   ```python
   form = TopicForm(request.POST, request.FILES)
   ```

## 7. Ответы на контрольные вопросы (ПР11)

1. **Для чего предназначены формы Django?**  
   Формы Django инкапсулируют логику отображения элементов интерфейса HTML (`<input>`, `<select>`), сбора и разбора пользовательских данных из HTTP-запроса, их очистки, комплексной валидации по типам и бизнес-правилам, а также вывода информативных сообщений об ошибках.

2. **Чем отличаются классы Form и ModelForm?**  
   `Form` — базовый универсальный класс для произвольных данных (поиск, фильтрация, калькуляторы), где каждое поле объявляется вручную. `ModelForm` жестко привязан к модели Django ORM (`Meta.model`), автоматически генерирует поля и виджеты на основе атрибутов модели и предоставляет готовый метод `.save()` для создания или обновления записей в БД.

3. **В чём разница между GET- и POST-запросами при отправке формы?**  
   - `GET`: параметры передаются открыто в строке URL (`?q=django&sort=title`). Метод идемпотентен, безопасен для многократного вызова, результаты можно сохранять в закладки браузера (применяется для поиска и фильтрации).
   - `POST`: данные передаются в теле HTTP-запроса. Метод изменяет состояние сервера и базы данных (создание заявки, удаление, редактирование) и обязательно требует CSRF-защиты.

4. **Что означает свойство is_bound и в каких случаях оно используется?**  
   Свойство `is_bound` возвращает `True`, если форма связана с переданными данными (`data=request.POST` или `request.GET`), и `False`, если форма пустая (несвязанная). Только для связанной формы можно вызывать метод `is_valid()` и обращаться к `cleaned_data` или `errors`.

5. **Для чего вызывается метод is_valid()?**  
   Метод `is_valid()` запускает полный конвейер валидации формы: преобразование сырых строк в Python-типы, валидацию полей, вызов методов `clean_<field>()` и кросс-полевого метода `clean()`. Метод возвращает `True`, если данные корректны и словарь ошибок пуст.

6. **Что содержит атрибут cleaned_data и когда к нему можно обращаться?**  
   Атрибут `cleaned_data` представляет собой словарь очищенных, нормализованных и приведенных к соответствующим Python-типам данных (например, `datetime.date`, `int`, экземпляры моделей). Обращаться к нему можно только после успешного выполнения `form.is_valid()`.

7. **Для чего предназначен метод clean_<field>()? Чем он отличается от метода clean()?**  
   - `clean_<field>()`: валидирует конкретное поле формы (например, `clean_capacity()`). Возвращает проверенное значение этого поля.
   - `clean()`: вызывается после валидации всех отдельных полей и предназначен для комплексной кросс-полевой валидации, когда корректность зависит от сочетания нескольких полей (например, сравнение GPA студента с минимальным GPA выбранной темы).

8. **Как в Django реализуется проверка введённых пользователем данных? Для чего используется ValidationError?**  
   Проверка реализуется валидаторами модели, валидаторами полей формы и методами `clean()`. При обнаружении некорректных данных возбуждается исключение `forms.ValidationError("Текст ошибки")`, которое перехватывается Django и добавляется в коллекцию ошибок `form.errors`.

9. **Для чего применяются виджеты (widgets) и как они влияют на отображение полей формы?**  
   Виджет определяет HTML-представление поля при рендеринге страницы (`<input type="text">`, `<textarea>`, `<select>`) и способ извлечения данных. С помощью виджетов задаются CSS-классы Bootstrap (`class="form-control"`), плейсхолдеры, атрибуты `min`/`max` и режим множественного выбора.

10. **Как связать ModelForm с моделью Django и ограничить набор полей, доступных для редактирования?**  
    Вложенный класс `Meta` связывает форму с моделью через атрибут `model = Topic`, а ограничение набора полей задается списком `fields = ["title", "capacity", "is_available"]` или `exclude = ["slug"]`.

11. **В чём разница между созданием и редактированием объекта с помощью ModelForm? Для чего при редактировании используется параметр instance?**  
    При создании форма инициализируется без существующего объекта (`form = ApplicationForm(request.POST)`), и метод `.save()` выполняет операцию `INSERT`. При редактировании в форму передается существующий экземпляр (`form = ApplicationEditForm(request.POST, instance=app)`): форма предварительно заполняется данными объекта, а `.save()` выполняет операцию `UPDATE`.

12. **Как организовать обработку формы в view-функции при GET- и POST-запросах?**  
    При `request.method == "POST"` форма связывается с данными `form = Form(request.POST)`. Если `form.is_valid()`, данные сохраняются и выполняется `redirect` (паттерн Post/Redirect/Get). При `request.method == "GET"` создается пустая форма `form = Form()`, после чего вызывается `render(request, "template.html", {"form": form})`.

13. **Для чего используется метод save() формы ModelForm?**  
    Метод `.save()` создает или обновляет экземпляр связанной модели на основе очищенных данных `cleaned_data` и сохраняет его в базе данных. При передаче `commit=False` метод возвращает объект модели без немедленной записи в БД, позволяя дозаполнить служебные поля до сохранения.

14. **Как реализовать проверку доступности темы (помещения) на выбранную дату с учётом существующих активных заявок?**  
    В методе `clean()` формы выполняется запрос:  
    `active_count = Application.objects.filter(topic=topic, status=Application.Status.ACTIVE).exclude(pk=self.instance.pk).count()`.  
    Если `active_count >= topic.capacity`, возбуждается `ValidationError("Квота мест исчерпана")`.

15. **Почему при отмене заявки (бронирования) следует изменять ее статус, а не удалять запись из базы данных?**  
    Паттерн мягкого удаления (Soft Delete) и смена статуса сохраняют юридическую, хронологическую и статистическую целостность истории учебного процесса кафедры, предотвращая нарушение внешних ключей и позволяя анализировать статистику отказов.

16. **Для чего используется request.FILES и какую роль играет атрибут enctype="multipart/form-data" при отправке файлов?**  
    Файлы, передаваемые в HTTP POST-запросе, не попадают в `request.POST`, а помещаются в словарь `request.FILES`. Атрибут `enctype="multipart/form-data"` в теге `<form>` указывает браузеру кодировать данные запроса частями (MIME-блоки), что является обязательным стандартом для передачи бинарных файлов на сервер.

17. **Как защитить POST-запросы от межсайтовой подделки запросов (CSRF) в Django?**  
    Защита обеспечивается встроенным `django.middleware.csrf.CsrfViewMiddleware`. В каждый HTML-шаблон формы внутри тега `<form>` помещается тег `{% csrf_token %}`, который генерирует скрытое поле с уникальным криптографическим токеном сессии, валидируемым сервером при отправке.

18. **Как организовать переход к редактированию конкретной заявки и обработать ситуацию, когда объект не найден?**  
    В URL-маршруте объявляется конвертер с первичным ключом `path("applications/<int:pk>/edit/", ...)`. В контроллере извлечение выполняется через `app = get_object_or_404(Application, pk=pk)`. При отсутствии объекта Django автоматически вернет страницу с кодом 404 «Объект не найден».
