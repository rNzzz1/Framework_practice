## 4. Реализация слоя данных (Django ORM)

### 4.1. Описание моделей в `topics/models.py`
```python
from datetime import date
from django.db import models
from django.core.exceptions import ValidationError


class Discipline(models.Model):
    name = models.CharField(max_length=200, unique=True, verbose_name="Наименование дисциплины")
    code = models.CharField(max_length=50, blank=True, verbose_name="Код дисциплины")
    department = models.CharField(max_length=200, default="Кафедра индустриального программирования", verbose_name="Кафедра")

    class Meta:
        verbose_name = "Дисциплина"
        verbose_name_plural = "Дисциплины"
        ordering = ["name"]

    def __str__(self) -> str:
        return self.name


class Student(models.Model):
    name = models.CharField(max_length=150, verbose_name="ФИО студента")
    course = models.PositiveSmallIntegerField(default=3, verbose_name="Курс обучения")
    gpa = models.FloatField(default=4.0, verbose_name="Средний балл (GPA)")
    is_budget = models.BooleanField(default=True, verbose_name="Бюджетная форма обучения")
    email = models.EmailField(unique=True, verbose_name="Электронная почта")

    class Meta:
        verbose_name = "Студент"
        verbose_name_plural = "Студенты"
        ordering = ["name"]

    def calculate_priority(self) -> int:
        base = int(float(self.gpa) * 20)
        bonus = 10 if self.is_budget else 0
        return base + bonus

    def __str__(self) -> str:
        budget_str = "Бюджет" if self.is_budget else "Договор"
        return f"{self.name} ({self.course} курс, GPA: {self.gpa:.2f}, {budget_str})"


class Topic(models.Model):
    title = models.CharField(max_length=255, verbose_name="Название темы")
    discipline = models.ForeignKey(Discipline, on_delete=models.PROTECT, related_name="topics", verbose_name="Дисциплина")
    supervisor = models.CharField(max_length=150, verbose_name="Научный руководитель")
    min_course = models.PositiveSmallIntegerField(default=3, verbose_name="Минимальный курс")
    min_gpa = models.FloatField(default=3.5, verbose_name="Минимальный балл GPA")
    capacity = models.PositiveSmallIntegerField(default=1, verbose_name="Квота мест")
    current_students = models.PositiveSmallIntegerField(default=0, verbose_name="Количество записанных студентов")
    is_available = models.BooleanField(default=True, verbose_name="Доступна для выбора")

    class Meta:
        verbose_name = "Тема курсовой работы"
        verbose_name_plural = "Темы курсовых работ"
        ordering = ["title"]

    def is_suitable_for(self, student_course: int, gpa: float) -> bool:
        return (student_course >= self.min_course) and (gpa >= self.min_gpa)

    def has_free_capacity(self) -> bool:
        active_apps = self.applications.filter(status=Application.Status.ACTIVE).count()
        return active_apps < self.capacity

    def reserve_slot(self) -> bool:
        if not self.has_free_capacity() or not self.is_available:
            return False
        self.current_students += 1
        if self.current_students >= self.capacity:
            self.is_available = False
        self.save()
        return True

    def release_slot(self) -> None:
        self.current_students = max(0, self.current_students - 1)
        self.is_available = True
        self.save()

    def __str__(self) -> str:
        status_text = "Свободна" if (self.is_available and self.has_free_capacity()) else "Занята"
        return f"[{self.id}] {self.title} ({self.discipline.name}) — {status_text}"


class Application(models.Model):
    class Status(models.TextChoices):
        ACTIVE = "active", "Активна"
        CANCELLED = "cancelled", "Отменена"
        COMPLETED = "completed", "Завершена"

    topic = models.ForeignKey(Topic, on_delete=models.PROTECT, related_name="applications", verbose_name="Тема курсовой работы")
    student = models.ForeignKey(Student, on_delete=models.PROTECT, related_name="applications", verbose_name="Студент-соискатель")
    application_date = models.DateField(default=date.today, verbose_name="Дата подачи заявки")
    status = models.CharField(max_length=20, choices=Status.choices, default=Status.ACTIVE, verbose_name="Статус заявки")
    priority = models.IntegerField(default=0, verbose_name="Рейтинговый балл / приоритет")

    class Meta:
        verbose_name = "Заявка на курсовую работу"
        verbose_name_plural = "Заявки на курсовые работы"
        ordering = ["-application_date", "-id"]

    def cancel(self) -> None:
        if self.status != self.Status.CANCELLED:
            self.status = self.Status.CANCELLED
            self.save(update_fields=["status"])
            self.topic.release_slot()
```

## 7. Ответы на контрольные вопросы (ПР9)

1. **Что такое Django ORM и какую задачу он решает?**  
   Django ORM (Object-Relational Mapping) — объектно-реляционный преобразователь, связывающий таблицы реляционной базы данных с Python-классами. Он позволяет взаимодействовать с базой данных на уровне объектно-ориентированного кода (создание, фильтрация, объединение записей) без написания ручных SQL-запросов, обеспечивая независимость от конкретной СУБД, защиту от SQL-инъекций и типобезопасность.

2. **Как модель Django связана с таблицей реляционной базы данных?**  
   Класс модели, унаследованный от `models.Model`, отображается в отдельную таблицу БД. Атрибуты класса, объявленные как `models.Field`, становятся столбцами таблицы соответствующих типов данных (`VARCHAR`, `INTEGER`, `DATE` и т.д.), а экземпляры модели представляют конкретные строки этой таблицы.

3. **Как Django определяет первичный ключ модели, если он явно не объявлен?**  
   Если в модели нет поля с параметром `primary_key=True`, Django автоматически добавляет целочисленное автоинкрементное поле `id` (обычно `BigAutoField` или `AutoField` в зависимости от `DEFAULT_AUTO_FIELD` в `settings.py`).

4. **Чем CharField, TextField, IntegerField, PositiveIntegerField, EmailField и BooleanField отличаются друг от друга?**  
   - `CharField`: строка ограниченной длины с обязательным параметром `max_length` (`VARCHAR` в SQL).
   - `TextField`: многострочный текст неограниченной длины (`TEXT` в SQL).
   - `IntegerField`: стандартное 32-битное целое число со знаком.
   - `PositiveIntegerField`: неотрицательное целое число (валидирует значения $\ge 0$).
   - `EmailField`: строковое поле с встроенным валидатором корректности email-адреса.
   - `BooleanField`: логическое значение `True`/`False` (`BOOLEAN` или `TINYINT(1)` в SQL).

5. **Как в Django реализуется связь «многие к одному»?**  
   Связь «многие к одному» (many-to-one) реализуется с помощью `models.ForeignKey` на стороне подчиненной (дочерней) модели. Например, множество тем курсовых работ ссылаются на одну дисциплину, или множество заявок ссылаются на одного студента.

6. **Для чего используется ForeignKey?**  
   `ForeignKey` создает в таблице базы данных столбец внешнего ключа с ограничением ссылочной целостности, связывающий строку текущей таблицы с первичным ключом родительской таблицы, и предоставляет удобный интерфейс доступа к объекту в Python (`app.topic`).

7. **Что означают параметры on_delete и related_name?**  
   - `on_delete`: задает поведение при удалении родительского объекта (`models.CASCADE` — каскадное удаление, `models.PROTECT` — запрет удаления при наличии ссылок, `models.SET_NULL` — зануление внешнего ключа).
   - `related_name`: имя атрибута для обратного доступа от родительского объекта к коллекции дочерних записей (например, `discipline.topics.all()`).

8. **Чем поле модели room отличается от атрибута room_id?**  
   `room_id` — это скалярное числовое значение первичного ключа связанной записи в базе данных. Поле `room` — это дескриптор Django ORM, возвращающий полноценный связанный Python-объект модели `Room` (выполняя SQL-запрос при необходимости или беря объект из кэша).

9. **Как получить все записи модели, отфильтровать их и отсортировать?**  
   С помощью цепочки методов QuerySet: `Topic.objects.all().filter(is_available=True).order_by("-min_gpa")`. Запросы являются ленивыми и объединяются в один SQL-запрос.

10. **Чем методы get() и filter() отличаются друг от друга?**  
    `filter()` возвращает объект `QuerySet` (коллекцию), который может содержать 0, 1 или множество элементов. `get()` возвращает ровно один экземпляр модели; если запись не найдена, возбуждается исключение `DoesNotExist`, а если найдено более одной записи — `MultipleObjectsReturned`.

11. **Для чего применяются select_related() и values_list()?**  
    - `select_related()`: выполняет SQL `JOIN` для загрузки данных связанных объектов по связям `ForeignKey`/`OneToOne` в один запрос, устраняя проблему N+1 запросов.
    - `values_list()`: возвращает кортежи (или скаляры при `flat=True`) конкретных столбцов вместо создания полных экземпляров моделей, существенно экономя память и время выполнения.

12. **Чем создание миграций (makemigrations) отличается от их применения (migrate)?**  
    `makemigrations` анализирует изменения в коде `models.py` и генерирует Python-файл миграции (план изменений схемы). `migrate` считывает непримененные файлы миграций и выполняет реальные DDL-инструкции в базе данных (`CREATE TABLE`, `ALTER TABLE`), фиксируя факт применения в таблице `django_migrations`.

13. **Для чего используется команда sqlmigrate?**  
    Команда `python manage.py sqlmigrate <app_name> <migration_number>` выводит сырой SQL-код (DDL), который будет сгенерирован и выполнен при применении указанной миграции, без внесения изменений в БД.

14. **Почему нельзя без проверки применять миграции к базе, созданной до внедрения Django ORM?**  
    База, созданная сторонними скриптами, может содержать другие типы данных, отличные имена столбцов, несогласованные внешние ключи или отсутствующую таблицу `django_migrations`. Применение миграций вслепую может привести к ошибкам `Table already exists` или повреждению данных.

15. **Какие способы переноса данных в новую структуру базы данных можно использовать?**  
    1) Экспорт данных в промежуточный формат (JSON/CSV) с последующей загрузкой через фикстуры `loaddata` или скрипт. 2) Написание скрипта прямого переноса через `sqlite3` + Django ORM в транзакции. 3) Использование Django Data Migrations (`RunPython`).

16. **Чем objects.create() отличается от создания объекта с последующим вызовом save()?**  
    `Model.objects.create(**kwargs)` инстанциирует объект и сразу же вызывает `save()`, сохраняя запись в базе данных в одну строчку. Создание экземпляра `obj = Model(...)` сначала инициализирует объект только в оперативной памяти Python, позволяя модифицировать его атрибуты до явного вызова `obj.save()`.

17. **Как реализовать поиск свободных тем (помещений) средствами ORM?**  
    Через метод модели `has_free_capacity()` или ORM-запрос:  
    `Topic.objects.annotate(active_count=Count('applications', filter=Q(applications__status='active'))).filter(is_available=True, current_students__lt=F('capacity'))`.

18. **Почему отменённая заявка (бронирование) не должна блокировать тему (помещение)?**  
    Отмененная заявка юридически и организационно прекращает резервирование квоты места. Если учитывать отмененные записи при контроле доступности, система будет ошибочно считать квоту темы исчерпанной, блокируя запись других подходящих соискателей.

19. **Как можно получить количество связанных объектов с помощью ORM?**  
    - Для конкретного экземпляра: `topic.applications.count()`.
    - Для всей выборки сразу: через аннотацию `Discipline.objects.annotate(total_apps=Count('topics__applications'))`.

20. **Какие изменения необходимо внести в представления и шаблоны при переходе от словарей к объектам моделей?**  
    В представлениях вместо обращений по ключам словаря `item['title']` используется точечная нотация `item.title`. В шаблонах синтаксис `{{ item.title }}` унифицирован, однако становятся доступными методы модели `{{ topic.has_free_capacity }}` и свойства связей `{{ app.topic.discipline.name }}` без предварительных ручных JOIN в коде view.

21. **В каких случаях удаление объекта может быть запрещено из-за связанных записей?**  
    Когда внешний ключ объявлен с параметром `on_delete=models.PROTECT`. При попытке вызова `.delete()` Django проверяет наличие связанных дочерних записей и возбуждает исключение `models.ProtectedError`, защищая исторические данные от потери ссылочной целостности.

22. **Почему необходимо проверять не только внешний вид приложения, но и исходный код?**  
    Корректный внешний вид HTML-страницы не гарантирует отсутствие скрытых критических дефектов: проблемы N+1 SQL-запросов, утечек памяти, отсутствия валидации диапазонов, некорректной обработки транзакций и уязвимостей SQL Injection или CSRF.

23. **Какие сведения необходимо обновить в README.md после перехода к ORM?**  
    Сведения об используемых реляционных моделях и их полях, структуру миграций, команды создания БД и выполнения миграций (`migrate`, `seed_data`), описание способов демонстрации запросов ORM и инструкцию по запуску тестов.

24. **Какие преимущества и ограничения есть у ORM по сравнению с выполнением SQL-запросов непосредственно через sqlite3?**  
    - **Преимущества:** высокая скорость разработки, независимость от диалекта SQL конкретной СУБД, автоматическая защита от SQL-инъекций, поддержка миграций схемы, кэширование, удобная валидация моделей.
    - **Ограничения:** небольшой оверхед по производительности на создание Python-объектов, сложность составления сверхсложных аналитических отчетов со специфическими оконными функциями без вызова `raw()`.
