### 4.1. Конфигурация административной панели (`topics/admin.py`)
```python
from django.contrib import admin
from django.utils.html import format_html
from .models import Discipline, Student, Topic, Application

# Кастомизация заголовков административного интерфейса РТУ МИРЭА
admin.site.site_header = "Панель управления сервисом курсовых работ РТУ МИРЭА"
admin.site.site_title = "Курсовые работы ИПТИП"
admin.site.index_title = "Администрирование учебного процесса и тем курсовых работ"


class ApplicationTabularInline(admin.TabularInline):
    """Встроенное табличное отображение заявок внутри карточек тем и студентов."""
    model = Application
    extra = 0
    fields = ("student", "topic", "application_date", "status", "priority")
    readonly_fields = ("application_date", "priority")
    show_change_link = True
    can_delete = False


@admin.register(Discipline)
class DisciplineAdmin(admin.ModelAdmin):
    list_display = ("id", "name", "code", "department", "topics_count")
    list_display_links = ("id", "name")
    search_fields = ("name", "code", "department")
    ordering = ("name",)
    list_per_page = 20

    @admin.display(description="Кол-во тем")
    def topics_count(self, obj: Discipline) -> int:
        return obj.topics.count()


@admin.register(Student)
class StudentAdmin(admin.ModelAdmin):
    list_display = (
        "id", "name", "course", "gpa", "is_honor_student",
        "is_budget", "email", "applications_count",
    )
    list_display_links = ("id", "name")
    list_filter = ("course", "is_budget")
    search_fields = ("name", "email")
    ordering = ("name",)
    list_per_page = 20
    readonly_fields = ("id", "is_honor_student")
    inlines = [ApplicationTabularInline]

    @admin.display(description="Подано заявок")
    def applications_count(self, obj: Student) -> int:
        return obj.applications.count()


@admin.register(Topic)
class TopicAdmin(admin.ModelAdmin):
    list_display = (
        "id", "title", "discipline", "supervisor", "capacity",
        "current_students", "is_available", "is_competitive", "site_link",
    )
    list_display_links = ("id", "title")
    list_filter = ("discipline", "min_course", "is_available")
    search_fields = ("title", "supervisor", "discipline__name")
    list_editable = ("capacity",)
    ordering = ("title",)
    list_per_page = 20
    readonly_fields = ("id", "slug", "current_students", "is_competitive")
    autocomplete_fields = ("discipline",)
    inlines = [ApplicationTabularInline]
    view_on_site = True

    @admin.display(description="Ссылка на сайте")
    def site_link(self, obj: Topic) -> str:
        url = obj.get_absolute_url()
        return format_html('<a href="{}" target="_blank">Просмотр ↗</a>', url)


@admin.register(Application)
class ApplicationAdmin(admin.ModelAdmin):
    list_display = (
        "id", "student", "topic", "application_date",
        "status", "priority", "status_badge",
    )
    list_display_links = ("id", "student")
    list_filter = ("status", "application_date", "topic__discipline")
    search_fields = ("student__name", "topic__title", "student__email")
    ordering = ("-application_date", "-id")
    date_hierarchy = "application_date"
    list_select_related = ("student", "topic", "topic__discipline")
    autocomplete_fields = ("student", "topic")
    readonly_fields = ("id", "priority")
    list_per_page = 20
    actions = ["cancel_applications", "restore_applications"]

    @admin.action(description="Отменить выбранные заявки")
    def cancel_applications(self, request, queryset):
        count = 0
        for app in queryset:
            if app.status != Application.Status.CANCELLED:
                app.cancel()
                count += 1
        self.message_user(request, f"Успешно аннулировано {count} заявок с освобождением квот.")

    @admin.action(description="Восстановить (активировать) выбранные заявки")
    def restore_applications(self, request, queryset):
        count = 0
        for app in queryset:
            if app.status != Application.Status.ACTIVE:
                app.status = Application.Status.ACTIVE
                app.save(update_fields=["status"])
                app.topic.reserve_slot()
                count += 1
        self.message_user(request, f"Восстановлено в статус активных: {count} заявок.")

    @admin.display(description="Индикатор статуса")
    def status_badge(self, obj: Application) -> str:
        colors = {
            Application.Status.ACTIVE: "#2ea44f",
            Application.Status.CANCELLED: "#cf222e",
            Application.Status.COMPLETED: "#0969da",
        }
        color = colors.get(obj.status, "#656d76")
        return format_html(
            '<span style="background-color: {}; color: white; padding: 3px 8px; '
            'border-radius: 10px; font-size: 11px; font-weight: bold;">{}</span>',
            color,
            obj.get_status_display(),
        )
```

# GeneratedField для вычисления статуса отличника на уровне базы данных:
is_honor_student = models.GeneratedField(
    expression=models.Case(
        models.When(gpa__gte=4.5, then=models.Value(True)),
        default=models.Value(False),
    ),
    output_field=models.BooleanField(),
    db_persist=True,
    verbose_name="Студент-отличник",
)

# GeneratedField для вычисления командной темы:
is_competitive = models.GeneratedField(
    expression=models.Case(
        models.When(capacity__gt=1, then=models.Value(True)),
        default=models.Value(False),
    ),
    output_field=models.BooleanField(),
    db_persist=True,
    verbose_name="Командная/Конкурсная тема",
)

# Кастомный QuerySet и Manager заявок:
class ApplicationQuerySet(models.QuerySet):
    def active(self):
        return self.filter(status=Application.Status.ACTIVE)

    def cancelled(self):
        return self.filter(status=Application.Status.CANCELLED)

    def completed(self):
        return self.filter(status=Application.Status.COMPLETED)

    def for_date(self, target_date):
        return self.filter(application_date=target_date)

class ApplicationManager(models.Manager.from_queryset(ApplicationQuerySet)):

from .models import Topic


def topic_context(request):
    """Поставляет глобальные переменные информационной системы во все шаблоны."""
    try:
        topics_total = Topic.objects.count()
        active_topics = Topic.objects.filter(is_available=True).count()

## 5. Анализ SQL-запросов и Django Debug Toolbar
Для мониторинга и исключения проблемы N+1 запросов подключен пакет `django-debug-toolbar`.
1. **Список тем и заявок:** при обращении к `/applications/` благодаря использованию:
   ```python
   Application.objects.select_related("topic", "student", "topic__discipline").all()
   ```
   выполняется ровно **1 SQL-запрос** с объединением `LEFT OUTER JOIN` вместо 1 + 3N запросов.

## 7. Ответы на контрольные вопросы (ПР10)

1. **Для чего предназначена административная панель Django?**  
   Административная панель Django — это готовый высокоуровневый веб-интерфейс для доверенных сотрудников и администраторов системы, предназначенный для просмотра, фильтрации, добавления, редактирования и удаления записей базы данных без необходимости создания индивидуальных страниц под каждую сущность.

2. **Как зарегистрировать модель в административной панели?**  
   Через вызов `admin.site.register(Model, ModelAdmin)` или с помощью декоратора над классом конфигурации: `@admin.register(Model) class MyAdmin(admin.ModelAdmin): pass`.

3. **Для чего используется класс admin.ModelAdmin?**  
   `ModelAdmin` инкапсулирует все правила и параметры отображения конкретной модели в интерфейсе админки: настраивает видимые столбцы, панели фильтрации, поисковые поля, встроенные формы связей, права доступа и кастомные групповые действия.

4. **Как с помощью list_display настроить столбцы списка записей?**  
   В кортеже `list_display` перечисляются имена полей модели или методов администратора, которые выводятся в таблице объектов. Метод может возвращать вычисленные значения или HTML-код (оформленный через `format_html`).

5. **Чем search_fields отличается от list_filter?**  
   `search_fields` предоставляет поле текстового поиска по подстроке (`LIKE`/`ILIKE`) среди указанных столбцов. `list_filter` формирует боковую панель дискретных фильтров (по фиксированным значениям, датам, внешним ключам или булевым флагам).

6. **Для чего используются ordering и list_per_page?**  
   - `ordering`: задает порядок сортировки записей по умолчанию (например, `("-application_date",)` — сначала новые).
   - `list_per_page`: ограничивает количество записей, выводимых на одной странице таблицы админки (по умолчанию 100; в проекте установлено 20).

7. **Как настроить отображение связанных объектов с помощью inline?**  
   Создается класс, унаследованный от `admin.TabularInline` или `admin.StackedInline`, в котором указывается дочерняя модель (`model = Application`). Затем этот класс подключается в атрибут `inlines = [ApplicationTabularInline]` родительского `ModelAdmin` (например, `TopicAdmin` или `StudentAdmin`).

8. **Что такое QuerySet и в какой момент ORM-запрос обычно обращается к базе данных?**  
   `QuerySet` — это ленивая коллекция объектов модели из базы данных. Запрос к СУБД отправляется только в момент материализации (evaluation): итерации в цикле `for`, среза с шагом, вызова функций `len()`, `list()`, `bool()`, `repr()` или методов `get()`, `count()`, `first()`.

9. **Чем отличаются методы filter() и exclude()?**  
   `filter()` включает в выборку только те записи, которые удовлетворяют заданным условиям (`WHERE condition`). `exclude()` отсекает записи, удовлетворяющие условиям, формируя отрицание (`WHERE NOT (condition)`).

10. **Для чего используется объект Q?**  
    Объект `django.db.models.Q` используется для построения сложных SQL-условий с логическими операторами ИЛИ (`|`), И (`&`) и НЕ (`~`), которые невозможно выразить обычными именованными аргументами в методе `.filter()`.

11. **Как с помощью ORM отсортировать записи и ограничить выборку?**  
    Сортировка выполняется методом `.order_by("field_name")` (или `"-field_name"` для убывания). Ограничение выборки выполняется стандартным Python-срезом QuerySet: `Model.objects.order_by("-gpa")[:5]` транслируется в SQL `ORDER BY gpa DESC LIMIT 5`.

12. **Чем отличаются get() и filter() при получении объектов?**  
    `filter()` возвращает `QuerySet`, даже если найдена одна запись или ничего не найдено (пустой список). `get()` возвращает ровно один объект модели; если записей 0 или больше 1, возбуждаются исключения `DoesNotExist` и `MultipleObjectsReturned`.

13. **Для чего используется get_object_or_404() в представлении Django?**  
    `get_object_or_404()` пытается получить объект модели с помощью `get()`. Если объект отсутствует, вместо падения сервера с необработанной системной ошибкой `DoesNotExist` (код 500) функция возбуждает стандартное исключение `django.http.Http404`, отображая пользователю страницу 404 «Не найдено».

14. **Чем отличаются select_related() и prefetch_related()?**  
    - `select_related()`: используется для связей «один к одному» и «многие к одному» (`ForeignKey`), выполняя один SQL-запрос с `JOIN`.
    - `prefetch_related()`: используется для связей «ко многим» (`ManyToManyField`, обратные связи `ForeignKey`), выполняя отдельный SQL-запрос с последующим связыванием объектов на уровне Python (`IN (...)`).

15. **Как с помощью ORM получить поля связанной модели?**  
    Через синтаксис двойного подчеркивания: при фильтрации `Application.objects.filter(topic__discipline__name="Фреймворки")` или при доступе к атрибутам экземпляра `app.topic.discipline.name`.

16. **Почему при работе со связанными объектами может возникать проблема N+1 запросов?**  
    Если при итерации по списку из N объектов обращаться к их связанным полям `ForeignKey`, Django по умолчанию для каждого из N объектов выполняет отдельный вторичный SQL-запрос. В результате вместо 1 общего запроса система выполняет 1 + N запросов, что катастрофически снижает производительность.

17. **Как проверить корректность фильтрации и сортировки в приложении?**  
    С помощью модульных тестов с проверкой порядка элементов в выборке (`self.assertEqual([x.id for x in res], [1, 2])`), профилирования в `django-debug-toolbar`, а также ручной верификации соответствия выводимых записей выбранным параметрам.

18. **Почему отменённые заявки (бронирования) не должны учитываться при определении занятости темы и подсчёте активных заявок?**  
    Отмена заявки освобождает место по квоте темы. Если учитывать аннулированные заявки, счетчик занятых мест будет завышен, что необоснованно закроет доступ к теме для других студентов.

19. **Почему настройка фильтров в Django Admin не заменяет проверку бизнес-правил?**  
    Административная панель — это инструмент прямого управления данными, а фильтры в ней лишь ограничивают отображение. Проверка бизнес-правил (недопустимость превышения квоты, проверка академических критериев, запрет дубликатов) должна быть строго реализована в методах моделей (`clean()`, `save()`, `reserve_slot()`) и формах валидации.

20. **Какие проверки необходимо выполнить перед фиксацией изменений в Git?**  
    Необходимо выполнить: 1) `python manage.py check` (проверка конфигурации); 2) `python manage.py showmigrations` (проверка миграций); 3) `python manage.py test` и `pytest -v` (прохождение всех тестов); 4) `flake8 .` (соответствие кода PEP 8); 5) `git status` и `git diff` (проверка отсутствия временных файлов и корректности изменений).

В рамках практической работы № 10 индивидуальный проект «Сервис выбора тем курсовых работ» получил развитую панель управления на базе Django Admin. 

Реализованы специализированные классы `ModelAdmin` с настройкой `list_display`, `list_filter`, автодополнением, поиском и инлайнами. Внедрены кастомные действия администратора для управления статусами заявок, перечисление `Status(TextChoices)`, вычисляемые на уровне СУБД поля `GeneratedField`, уникальные канонические слаги тем с автотранслитерацией `unidecode`, глобальный контекстный процессор и инструмент профилирования SQL-запросов `django-debug-toolbar`.
